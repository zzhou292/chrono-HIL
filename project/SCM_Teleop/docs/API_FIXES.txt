C++ ChTireTestRig Reference (for future implementation):
--------------------------------------------------------
Proper usage pattern from demo_VEH_TireTestRig.cpp:
  
  auto wheel = ReadWheelJSON("vehicle/hmmwv/wheel/HMMWV_Wheel.json");
  auto tire = ReadTireJSON("vehicle/hmmwv/tire/HMMWV_TMeasyTire.json");
  ChTireTestRig rig(wheel, tire, sys);
  
  rig.SetNormalLoad(3000);  // Normal force in Newtons
  rig.SetTireStepsize(step_size);
  
  // Set SCM terrain
  ChTireTestRig::TerrainParamsSCM params;
  params.Bekker_Kphi = 2e6;      // Frictional modulus
  params.Bekker_Kc = 0;          // Cohesive modulus
  params.Bekker_n = 1.1;         // Exponent
  params.Mohr_cohesion = 0;      // Cohesion (Pa)
  params.Mohr_friction = 30;     // Friction angle (degrees)
  params.Janosi_shear = 0.01;    // Shear parameter (m)
  params.grid_spacing = 0.05;    // Grid resolution
  rig.SetTerrainSCM(size, params);
  
  // Set motion - Option 1: Specify slip ratio directly
  rig.SetConstantLongitudinalSlip(slip_ratio, base_speed);
  rig.SetSlipAngleFunction(chrono_types::make_shared<ChFunctionConst>(slip_angle));
  
  // Set motion - Option 2: Specify speeds independently
  rig.SetLongSpeedFunction(chrono_types::make_shared<ChFunctionConst>(0.2));
  rig.SetAngSpeedFunction(chrono_types::make_shared<ChFunctionConst>(10 * CH_RPM_TO_RAD_S));
  
  rig.SetTimeDelay(1.0);  // Settling time before measurements
  rig.Initialize(ChTireTestRig::Mode::TEST);
  
  // Simulation loop
  while (running) {
      rig.Advance(step_size);
      
      auto tforce = rig.ReportTireForce();  // Returns TerrainForce struct
      double fx = tforce.force.x();
      double fy = tforce.force.y();
      double fz = tforce.force.z();
      
      double long_slip = rig.GetLongitudinalSlip();
      double slip_angle = rig.GetSlipAngle();
  }
